import numpy as np
import matplotlib.pyplot as plt

# Step 1: Activation Function
def step_function(x):
    if x >= 0:
        return 1
    else:
        return 0

# Step 2: Create Perceptron Class
class Perceptron:

    def __init__(self, input_size, learning_rate=0.1, epochs=10):
        self.weights = np.zeros(input_size + 1)
        self.lr = learning_rate
        self.epochs = epochs

    # Prediction
    def predict(self, x):
        x = np.insert(x, 0, 1)
        weighted_sum = np.dot(self.weights, x)
        return step_function(weighted_sum)

    # Training
    def train(self, X, y):
        for epoch in range(self.epochs):
            for xi, target in zip(X, y):
                xi = np.insert(xi, 0, 1)
                output = step_function(np.dot(self.weights, xi))
                error = target - output
                self.weights += self.lr * error * xi

            print(f"Epoch {epoch+1}: Weights = {self.weights}")

# Step 3: AND Gate Dataset
X = np.array([
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
])

y = np.array([0, 0, 0, 1])

print("AND Gate Input:")
print(X)

print("\nTarget Output:")
print(y)

# Step 4: Create and Train Perceptron
perceptron = Perceptron(input_size=2,
                        learning_rate=0.1,
                        epochs=10)

print("\nTraining Perceptron:\n")
perceptron.train(X, y)

# Step 5: Display Predictions
print("\nAND Gate Predictions:\n")

for x in X:
    prediction = perceptron.predict(x)
    print(f"Input: {x} --> Output: {prediction}")

# Step 6: Visualize Decision Boundary
x_min, x_max = -0.5, 1.5
y_min, y_max = -0.5, 1.5

xx, yy = np.meshgrid(
    np.arange(x_min, x_max, 0.01),
    np.arange(y_min, y_max, 0.01)
)

Z = []

for point in np.c_[xx.ravel(), yy.ravel()]:
    Z.append(perceptron.predict(point))

Z = np.array(Z).reshape(xx.shape)

plt.figure(figsize=(6, 5))

plt.contourf(xx, yy, Z, alpha=0.3, cmap="coolwarm")

plt.scatter(
    X[:, 0],
    X[:, 1],
    c=y,
    cmap="coolwarm",
    edgecolors="black",
    s=120
)

plt.xlabel("Input 1")
plt.ylabel("Input 2")
plt.title("Perceptron Decision Boundary (AND Gate)")
plt.grid(True)
plt.show()

# step I : Import libraries
import numpy as np
import matplotlib.pyplot as plt

# step II: create datasets
x = np.array([[0,0],[0,1],[1,0],[1,1]])
y = np.array([0,0,0,1])

# step III: Initialize weights,bias,learning rate, epochs
weights = np.zeros(2)
bias = 0
lr = 0.1 
epochs = 10

# step IV: Define the Activation Function
def activation(z):
    if z>=0:
        return 1
    return 0

# step V: Training loops
for epoch in range(epochs):
    # loop through all training examples
    for i in range(4):
        # apply forumula 
        z = np.dot(x[i],weights) + bias
        # predict
        y_pred = activation(z)
        # calculate error 
        error = y[i] - y_pred;

        # update the weights and bias
        weights = weights + lr*error*x[i]
        bias = bias + lr*error
    print("Epoch : ",epoch+1, weights,bias)

# Step VI: Testing 
for i in range(4):
    z = np.dot(x[i],weights) + bias
    y_pred = activation(z)
    print(x[i],y_pred)
