# Step 1: Import Libraries
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
import time

# Step 2: Load Local MNIST Dataset
data = np.load("mnist.npz")

x_train = data["x_train"].reshape(-1, 784).astype("float32") / 255.0
y_train = data["y_train"]

x_test = data["x_test"].reshape(-1, 784).astype("float32") / 255.0
y_test = data["y_test"]

# Step 3: Define Activation Functions
activations = ['sigmoid', 'tanh', 'relu']
results = {}
gradients_result = {}

# Step 4: Train Models and Calculate Gradients
for act in activations:

    model = tf.keras.Sequential([
        tf.keras.layers.Input(shape=(784,)),
        tf.keras.layers.Dense(128, activation=act),
        tf.keras.layers.Dense(64, activation=act),
        tf.keras.layers.Dense(10, activation='softmax')
    ])

    model.compile(optimizer='adam',
                  loss='sparse_categorical_crossentropy',
                  metrics=['accuracy'])

    start = time.time()

    history = model.fit(x_train, y_train, epochs=5,
                        batch_size=128, validation_split=0.2,
                        verbose=1)

    training_time = time.time() - start
    loss, accuracy = model.evaluate(x_test, y_test, verbose=0)

    results[act] = [model, history.history, training_time, accuracy]

    # Calculate Gradients
    with tf.GradientTape() as tape:
        predictions = model(x_train[:128], training=True)
        loss = tf.reduce_mean(
            tf.keras.losses.sparse_categorical_crossentropy(
                y_train[:128], predictions
            )
        )

    grads = tape.gradient(loss, model.trainable_variables)

    gradients_result[act] = [
        tf.reduce_mean(tf.abs(g)).numpy()
        for g in grads if g is not None
    ]

    print(act, "Accuracy:", accuracy * 100, "%")
    print("Training Time:", training_time, "seconds")


# Step 5: Plot Accuracy and Loss
for metric in ['accuracy', 'loss']:
    for act in activations:
        plt.plot(results[act][1][metric], label=act)

    plt.title(metric.capitalize() + " Comparison")
    plt.xlabel("Epochs")
    plt.ylabel(metric.capitalize())
    plt.legend()
    plt.grid()
    plt.show()


# Step 6: Plot Training Time
plt.bar(activations, [results[a][2] for a in activations])
plt.title("Training Time Comparison")
plt.xlabel("Activation Function")
plt.ylabel("Time (seconds)")
plt.show()


# Step 7: Plot Gradient Flow
for act in activations:
    plt.plot(gradients_result[act], marker='o', label=act)

plt.title("Gradient Flow Comparison")
plt.xlabel("Trainable Parameter Tensors")
plt.ylabel("Mean Absolute Gradient")
plt.legend()
plt.grid()
plt.show()


# Step 8: Display Final Results
print("\nFINAL COMPARISON")

for act in activations:
    print("\nActivation:", act)
    print("Test Accuracy:", results[act][3] * 100, "%")
    print("Training Time:", results[act][2], "seconds")
    print("Gradient Values:", gradients_result[act])

import struct

# Step 2: Load Local MNIST Dataset

def load_images(file):
    with open(file, 'rb') as f:
        magic, num, rows, cols = struct.unpack('>IIII', f.read(16))
        images = np.frombuffer(f.read(), dtype=np.uint8)
    return images.reshape(num, rows * cols)

def load_labels(file):
    with open(file, 'rb') as f:
        magic, num = struct.unpack('>II', f.read(8))
        labels = np.frombuffer(f.read(), dtype=np.uint8)
    return labels

x_train = load_images('MNIST/train-images-idx3-ubyte').astype('float32') / 255.0

print("Testing Data:", x_test.shape)
